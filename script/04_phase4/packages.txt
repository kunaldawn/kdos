# Firmware. Without it a real machine has no Wi-Fi, no Bluetooth and — this is
# the one that surprises people — no GPU: from Vega/GFX9 on, amdgpu must load
# PSP/SMU/GFX/SDMA blobs before it can bring the display up, and Xe and Gen12+
# i915 refuse to load without GuC. You fall through to simpledrm on the EFI
# framebuffer: a picture, no EGL, no compositor.
#
# Vendored, not upstream's 1.9 GB tree — ports/core/linux-firmware/vendor.py
# records the curated set and what was dropped and why.
linux-firmware

# The clock, the logs and the manual. Absent until wave 0: a skewed clock
# breaks TLS (so ports/fetch, git and every alien app), every daemon's
# diagnostics went to /dev/null, and the ISO carried thousands of man pages
# with nothing able to read them.

# Periodic jobs. snooze is one process per job under ksvc rather than a cron
# daemon with its own scheduler — it fits the supervisor KDOS already has.

# mDNS, and the only reason it is here: CUPS' driverless printer discovery is
# compiled out without it. musl has no NSS module system, so `.local` name
# resolution will never work — avahi's own API is what CUPS uses.

avahi

# X11 compatibility (Xwayland only — no Xorg server)
# ---------------------------------
# Alien apps that are X11-only (GIMP 2.x, Wine, older IDEs) need an X server.

# Icon-theme skeleton: the index.theme that makes /usr/share/icons/hicolor a
# real theme. Qt/Quickshell icon lookup silently fails without it — every app
# icon renders as the missing-texture checkerboard.
hicolor-icon-theme

# Boot splash. Lives in src/packages rather than ports/core because the source
# is ours, not an upstream tarball. Draws the CRT power-on animation straight
# to /dev/fb0 from the initramfs and keeps running until rcS hands the console
# to agetty.
kdos-splash

# Service supervisor and the console getty wrapper (src/packages, ours).
# Replaces fs/usr/sbin/service and fs/usr/local/sbin/kdos-getty, which is why
# neither is in fs/ any more.
kdos-tools

# The GTK/icon/cursor generators (src/packages, ours). Build-time step for the
# three art packages below AND the runtime engine behind `kdos theme` — which
# is why python3 is no longer a runtime dependency of the shipped system.
kdos-theme

# KDOS phosphor icon theme (src/packages, ours) — recolored Papirus.
kdos-icons

# KDOS phosphor pixel-art cursor theme (src/packages, ours) — generated Xcursor.

# KDOS phosphor GTK theme (src/packages, ours) — recolored adw-gtk3.

# Alien app runtime + appbox manager (src/packages, ours) — replaces the
