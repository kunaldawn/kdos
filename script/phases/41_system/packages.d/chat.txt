# MATRIX, WITH vim'S KEYS. SQLite is bundled because every rust port here is a
# static-pie binary and none of them can reach the shared libsqlite3 the image
# carries; TLS is rustls for the same reason pizauth's is.
iamb

# IRC AS A DIRECTORY. A FIFO to write into and a text file to read, per channel
# — so the client is `tail -f` and the composer is `echo`, and a bot is a shell
# script with no library at all. There is no other IRC client on this image.
ii

# libre's version is pinned to baresip's — a mismatch aborts at call time, not
# at link time.
libre

# The XMPP client library profanity links. A server nobody can connect to from
# the machine it runs on is a daemon, not a way to talk to somebody.
libstrophe

# OMEMO, the end-to-end encryption Dino speaks too.
libomemo-c

# IRC and the rest of the chat networks, in cells. The three scripting plugins
# are hard configure dependencies as well as runtime ones: dropping python3,
# perl or lua from the recipe's depends stops the build rather than narrowing
# it.
weechat

olm
mtxclient
