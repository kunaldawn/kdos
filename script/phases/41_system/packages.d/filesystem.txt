# The kernel mounts btrfs, xfs, f2fs, ntfs3 and exfat, and these are what

# A container format is one enormous file that changes on every write, so
# syncing it resends the whole thing and two writers corrupt it. gocryptfs
# keeps a ciphertext file per plaintext file, which is why it is the one that
# works inside syncthing.
gocryptfs

# samba is the INTEROPERABILITY answer (a Windows laptop, a phone or a printer
# speaks SMB and often nothing else); nfs-utils is the CORRECTNESS one — SMB
# does not carry POSIX ownership, symlinks or byte-range locks the way a Linux
# program expects. The v3 helpers are built and never started.
nfs-utils

# The two remote filesystems a workstation actually mounts. cifs-utils pulls
# talloc, which samba requires unconditionally.
cifs-utils
sshfs
