# Everything with no graphics and no toolkit in its dependency closure:
# services, networking, storage, the command line, codecs, and the science and
# hardware libraries.
#
# The files in packages.d are read in sorted order as one list. Together they
# name every port the phase installs and nothing it does not;
# testing/phaseclosure.py fails when a dependency pulls a port in from outside
# them, or when a shelf file names a port an earlier phase installs.

# The 1997 AAlib demo, hard-forked as ours: see src/art/kdos-bb/KDOS-FORK.
kdos-bb

# Boot splash. Lives in src/art rather than ports/core because the source is
# ours, not an upstream tarball. Draws straight to /dev/fb0 from the initramfs
# — a fade from black, or Phosphor mode's CRT power-on — and keeps running
# until rcS hands the screen to the display manager.
kdos-splash

# The identity generators (src/art, ours): KDE colour schemes, Konsole
# profiles, qt5ct palettes, wallpapers and the greeter's palette, all from
# libkcolor's one table, run at build time so python3 is not a runtime
# dependency of the shipped system.
kdos-theme

# The KDOS and KDOS-Light icon themes: the KDOS marks over Breeze, following
# the accent.
kdos-icons

# Plasma wallpaper packages: the KDOS mark in each accent, dark and light,
# drawn by kdos-theme at build time.
kdos-wallpapers
