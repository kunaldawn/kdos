# The files in packages.d are read in sorted order as one list. Together they

# testing/phaseclosure.py fails when a dependency pulls a port in from outside
# them, or when a shelf file names a port an earlier phase installs.

# git send-email's TLS and SMTP authentication are these perl modules, which it
# loads only when a send needs them: without them every send over TLS or with a
# login fails, and nothing else notices.

perl-digest-hmac
perl-crypt-urandom
perl-authen-sasl

perl-uri

perl-font-ttf
