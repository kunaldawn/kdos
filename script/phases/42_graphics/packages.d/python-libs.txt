# The compiled python bindings a document pipeline needs: an imaging library, a
# PDF rewriter and a HEIF decoder. Each binds a system library this tree
# already builds, which is why they are ports rather than entries in somebody's
# vendor bundle.
python3-pillow
python3-pikepdf
python3-pillow-heif

# WHAT visidata OPENS AN xlsx WITH. sc-im reads the format in C and needs none
# of this; visidata asks for openpyxl by name and refuses without it.
python3-openpyxl

# Binds pdfium for ocrmypdf.
python3-pypdfium2

# Shapes the text fpdf2 writes for ocrmypdf, against the system harfbuzz
# through python3-pkgconfig.
python3-uharfbuzz

python3-qrcode
